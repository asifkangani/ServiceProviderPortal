==================spp_vg_documents===================
CREATE TABLE `spp_vg_documents` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `org_id` bigint NOT NULL,
  `meta_document_id` bigint NOT NULL,
  `document_name` varchar(255) DEFAULT NULL,
  `document_data` longtext NOT NULL,
  `content_type` varchar(50) DEFAULT NULL,
  `created_on` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `fk_org` (`org_id`),
  KEY `fk_meta` (`meta_document_id`),
  CONSTRAINT `fk_meta` FOREIGN KEY (`meta_document_id`) REFERENCES `spp_vg_meta_documents` (`id`),
  CONSTRAINT `fk_org` FOREIGN KEY (`org_id`) REFERENCES `spp_vg_organization_details` (`id`)
);
